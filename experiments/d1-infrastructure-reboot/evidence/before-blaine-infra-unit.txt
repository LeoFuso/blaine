# /etc/systemd/system/blaine-infra.service
[Unit]
Description=Blaine local infrastructure Compose deployment
Requires=docker.service postgresql@18-main.service redis-server.service
After=docker.service postgresql@18-main.service redis-server.service network-online.target
PartOf=docker.service
RequiresMountsFor=/srv/blaine/infra

[Service]
Type=oneshot
RemainAfterExit=yes
WorkingDirectory=/etc/blaine/infra
ExecStart=/usr/bin/docker compose up --detach --wait --wait-timeout 300
ExecStop=/usr/bin/docker compose stop --timeout 60
ExecReload=/usr/bin/docker compose up --detach --wait --wait-timeout 300
TimeoutStartSec=600
TimeoutStopSec=300
UMask=0077

[Install]
WantedBy=multi-user.target
